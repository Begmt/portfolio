Учебные примеры проверки подписи, не готовые CRM-интеграции. Контракт подписи зависит от источника; для бизнес-операций нужна надёжная очередь до ответа 200.

// Учебный приёмник: контракт X-Signature = HMAC-SHA256 hex.
import { createServer } from 'node:http';
import { createHmac, timingSafeEqual } from 'node:crypto';
const secret = process.env.WEBHOOK_SECRET;
if (!secret) throw new Error('WEBHOOK_SECRET required');
createServer(async (req, res) => {
  if (req.method !== 'POST' || req.url !== '/webhook') {
    res.writeHead(404).end(); return;
  }
  const chunks = []; let size = 0;
  for await (const chunk of req) {
    size += chunk.length;
    if (size > 65536) { res.writeHead(413).end(); return; }
    chunks.push(chunk);
  }
  const body = Buffer.concat(chunks);
  const signature = req.headers['x-signature'];
  const expected = createHmac('sha256', secret).update(body).digest();
  if (typeof signature !== 'string' || !/^[a-f0-9]{64}$/i.test(signature)
      || !timingSafeEqual(Buffer.from(signature, 'hex'), expected)) {
    res.writeHead(403).end(); return;
  }
  // Только подтверждение приёма; бизнес-операции здесь не выполняются.
  res.writeHead(200, { 'Content-Type': 'text/plain' }).end('OK');
}).listen(Number(process.env.PORT || 3000));
# Учебный приёмник: X-Signature = HMAC-SHA256 hex.
import os, hmac, hashlib, re
from flask import Flask, request

app = Flask(__name__)
app.config['MAX_CONTENT_LENGTH'] = 65536
secret = os.environ['WEBHOOK_SECRET'].encode()

@app.post('/webhook')
def handle_webhook():
    body = request.get_data()
    signature = request.headers.get('X-Signature', '')
    expected = hmac.new(secret, body, hashlib.sha256).hexdigest()
    if not re.fullmatch(r'[a-fA-F0-9]{64}', signature) or not hmac.compare_digest(expected, signature.lower()):
        return 'Forbidden', 403
    # Только подтверждение приёма, без синхронных обращений к CRM.
    return 'OK', 200

// Учебный приёмник: X-Signature = HMAC-SHA256 hex.
$secret = getenv('WEBHOOK_SECRET');
if (!$secret) { http_response_code(503); exit; }
if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    http_response_code(405); exit;
